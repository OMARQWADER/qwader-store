                  <label
                    onClick={() => setSelectedMethod('authenticator')}
                    className={`p-4 rounded-2xl border flex items-center gap-3 cursor-pointer transition-all ${
                      selectedMethod === 'authenticator'
                        ? 'bg-purple-600/20 border-purple-500 text-white'
                        : 'bg-white/5 border-white/10 text-slate-300 hover:bg-white/10'
                    }`}
                  >
                    <Smartphone className="w-5 h-5 text-purple-400 shrink-0" />
                    <div>
                      <h5 className="text-xs font-bold font-cairo">
                        {language === 'ar' ? 'تطبيق المصادقة (Authenticator App)' : 'Authenticator App'}
                      </h5>
                      <p className="text-[11px] text-slate-400">Google Authenticator, Authy, 1Password</p>
                    </div>
                  </label>

                  <label
                    onClick={() => setSelectedMethod('whatsapp')}
                    className={`p-4 rounded-2xl border flex items-center gap-3 cursor-pointer transition-all ${
                      selectedMethod === 'whatsapp'
                        ? 'bg-purple-600/20 border-purple-500 text-white'
                        : 'bg-white/5 border-white/10 text-slate-300 hover:bg-white/10'
                    }`}
                  >
                    <Phone className="w-5 h-5 text-emerald-400 shrink-0" />
                    <div>
                      <h5 className="text-xs font-bold font-cairo">
                        {language === 'ar' ? 'رسائل الواتساب (WhatsApp OTP)' : 'WhatsApp OTP'}
                      </h5>
                      <p className="text-[11px] text-slate-400">
                        {language === 'ar' ? `إرسال الرمز إلى رقمك (${currentUser?.phone})` : `Send codes to ${currentUser?.phone}`}
                      </p>
                    </div>
                  </label>
                </div>

                <button
                  onClick={async () => {
                    if (selectedMethod === 'authenticator') {
                      setSetupStep('scan');
                    } else {
                      const sent = await sendSetupOtp();
                      if (sent) setSetupStep('verify');
                    }
                  }}
                  className="w-full py-3 rounded-full bg-purple-600 hover:bg-purple-500 text-white font-bold text-xs shadow-lg shadow-purple-900/30 transition-all font-cairo"
                >
                  {language === 'ar' ? 'المتابعة للخطوة التالية ➔' : 'Continue ➔'}
                </button>
              </div>
            )}

            {/* Step 2: QR Scan (for Authenticator) */}
            {setupStep === 'scan' && (
              <div className="space-y-4 text-center">
                <p className="text-xs text-slate-300">
                  {language === 'ar'
                    ? 'امسح رمز QR بواسطة تطبيق المصادقة (مثل Google Authenticator) أو ادخل المفتاح يدوياً:'
                    : 'Scan the QR code with your authenticator app or enter the key manually:'}
                </p>

                <div
                  className="p-4 bg-white rounded-2xl w-48 h-48 mx-auto flex items-center justify-center shadow-lg"
                  dangerouslySetInnerHTML={{
                    __html: generateQRCodeSVG(
                      getOTPAuthUri(tempSecret, currentUser?.email || 'user@qwaderstore.jo', 'QwaderStore')
                    ),
                  }}
                />

                <div className="p-3 rounded-xl bg-white/5 border border-white/10 flex items-center justify-between text-xs">
                  <span className="font-mono text-purple-300 font-bold select-all">{tempSecret}</span>
                  <button
                    onClick={() => {
                      navigator.clipboard?.writeText(tempSecret);
                      setCopiedSecret(true);
                      setTimeout(() => setCopiedSecret(false), 2000);
                    }}
                    className="text-slate-400 hover:text-white"
                  >
                    {copiedSecret ? <Check className="w-4 h-4 text-emerald-400" /> : <Copy className="w-4 h-4" />}
                  </button>
                </div>

                <button
                  onClick={() => setSetupStep('verify')}
                  className="w-full py-3 rounded-full bg-purple-600 hover:bg-purple-500 text-white font-bold text-xs transition-all font-cairo"
                >
                  {language === 'ar' ? 'لقد قمت بمسح الرمز، متابعة للتحقق ➔' : 'I scanned the code, proceed ➔'}
                </button>
              </div>
            )}

            {/* Step 3: Code Verification */}
            {setupStep === 'verify' && (
              <form onSubmit={handleConfirm2FAEnable} className="space-y-4 text-center">
                <p className="text-xs text-slate-300">
                  {selectedMethod === 'authenticator'
                    ? language === 'ar' ? 'أدخل الرمز المكون من 6 أرقام من تطبيق المصادقة لتأكيد التفعيل:' : 'Enter the 6-digit code generated by your app:'
                    : language === 'ar' ? `أدخل الرمز المكون من 6 أرقام المرسل عبر الواتساب:` : 'Enter the 6-digit code sent to your phone:'}
                </p>

                <input
                  type="text"
                  autoFocus
                  required
                  value={verifyCodeInput}
                  onChange={(e) => setVerifyCodeInput(e.target.value)}
                  placeholder="123456"
                  className="w-full p-3.5 rounded-2xl text-center text-xl tracking-widest font-mono font-bold bg-white/10 border border-purple-500/40 text-white focus:border-purple-400 focus:outline-none"
                />

                {setupError && (
                  <div className="p-3 rounded-xl bg-rose-500/20 border border-rose-500/40 text-rose-300 text-xs font-semibold">
                    {setupError}
                  </div>
                )}

                <button
                  type="submit"
                  className="w-full py-3 rounded-full bg-emerald-600 hover:bg-emerald-500 text-white font-bold text-xs shadow-lg shadow-emerald-900/30 transition-all font-cairo"
                >
                  {language === 'ar' ? 'تأكيد وتفعيل الحماية 🛡️' : 'Verify & Activate 🛡️'}
                </button>
              </form>
            )}

            {/* Step 4: Backup Codes Output */}
            {setupStep === 'backup' && (
              <div className="space-y-4 text-center">
                <div className="p-3 rounded-xl bg-emerald-500/20 border border-emerald-500/40 text-emerald-300 text-xs font-bold">
                  {language === 'ar' ? 'تم تفعيل التحقق بخطوتين بنجاح! 🎉' : '2FA activated successfully! 🎉'}
                </div>

                <p className="text-xs text-slate-300">
                  {language === 'ar'
                    ? 'احفظ رموز الاسترداد الاحتياطية التالية في مكان آمن جداً. ستحتاجها لو أضعت هاتفك:'
                    : 'Save these emergency recovery codes in a safe place. You will need them if you lose your phone:'}
                </p>

                <div className="grid grid-cols-2 gap-2 font-mono text-xs p-4 bg-black/50 rounded-2xl border border-white/10">
                  {tempBackupCodes.map((code, idx) => (
                    <div key={idx} className="p-2 bg-white/5 rounded-xl text-amber-300 font-bold">
                      {code}
                    </div>
                  ))}
                </div>

                <div className="flex gap-2">
                  <button
                    type="button"
                    onClick={handleDownloadBackupCodes}
                    className="flex-1 py-2.5 rounded-xl bg-white/10 hover:bg-white/20 text-slate-200 text-xs font-bold transition-all"
                  >
                    {language === 'ar' ? 'تحميل كملف' : 'Download TXT'}
                  </button>
                  <button
                    type="button"
                    onClick={() => setIs2FASetupModalOpen(false)}
                    className="flex-1 py-2.5 rounded-xl bg-purple-600 hover:bg-purple-500 text-white text-xs font-bold transition-all font-cairo"
                  >
                    {language === 'ar' ? 'إتمام وإغلاق' : 'Done & Close'}
                  </button>
                </div>
              </div>
            )}
          </div>
        </div>
      )}

      {/* MODAL 2: NEW CUSTOMER PROFILE COMPLETE */}
      {isProfileModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/80 backdrop-blur-md">
          <div className="w-full max-w-md p-6 sm:p-8 rounded-3xl glass-card border border-white/10 space-y-6">
            <div className="text-center">
              <h3 className="text-lg font-bold text-slate-100 font-cairo">
                {language === 'ar' ? 'استكمال بيانات الحساب الجديد' : 'Complete Your Profile'}
              </h3>
              <p className="text-xs text-slate-400 mt-1">
                {language === 'ar' ? 'يرجى إدخال اسمك ورقم هاتفك لإكمال تسجيل الحساب' : 'Enter your name and phone to finalize account creation'}
              </p>
            </div>

            <form onSubmit={handleProfileSubmit} className="space-y-4">
              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs text-slate-300 font-semibold mb-1">{language === 'ar' ? 'الاسم الأول' : 'First Name'}</label>
                  <input
                    type="text"
                    required
                    value={profileFirstName}
                    onChange={(e) => setProfileFirstName(e.target.value)}
                    className="w-full p-3 rounded-2xl text-xs bg-white/5 border border-white/10 text-white focus:border-purple-500 focus:outline-none"
                  />
                </div>
                <div>
                  <label className="block text-xs text-slate-300 font-semibold mb-1">{language === 'ar' ? 'اسم العائلة' : 'Last Name'}</label>
                  <input
                    type="text"
                    required
                    value={profileLastName}
                    onChange={(e) => setProfileLastName(e.target.value)}
                    className="w-full p-3 rounded-2xl text-xs bg-white/5 border border-white/10 text-white focus:border-purple-500 focus:outline-none"
                  />
                </div>
              </div>

              <div>
                <label className="block text-xs text-slate-300 font-semibold mb-1">{t.phone}</label>
                <div className="flex gap-2">
                  <select
                    value={profileCountryCode}
                    onChange={(e) => setProfileCountryCode(e.target.value)}
                    className="p-3 rounded-2xl text-xs bg-white/5 border border-white/10 text-white focus:border-purple-500 focus:outline-none"
                  >
                    <option value="+962" className="bg-slate-900">+962 (🇯🇴)</option>
                    <option value="+966" className="bg-slate-900">+966 (🇸🇦)</option>
                    <option value="+971" className="bg-slate-900">+971 (🇦🇪)</option>
                  </select>
                  <input
                    type="tel"
                    required
                    value={profilePhone}
                    onChange={(e) => setProfilePhone(e.target.value)}
                    placeholder="791234567"
                    className="flex-1 p-3 rounded-2xl text-xs bg-white/5 border border-white/10 text-white focus:border-purple-500 focus:outline-none"
                    dir="ltr"
                  />
                </div>
              </div>

              <button
                type="submit"
                className="w-full py-3.5 rounded-full bg-purple-600 hover:bg-purple-500 text-white font-bold text-xs shadow-lg shadow-purple-900/30 transition-all font-cairo"
              >
                {language === 'ar' ? 'حفظ وإنشاء الحساب 🚀' : 'Create Account 🚀'}
              </button>
            </form>
          </div>
        </div>
      )}
    </div>
  );
};
